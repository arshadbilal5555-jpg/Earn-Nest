async function requestWithdrawal() { const amountInput = document.getElementById('withdrawAmount'); const methodInput = document.getElementById('paymentMethod'); const accountInput = document.getElementById('paymentAccount'); const amount = Number(amountInput.value); const paymentMethod = methodInput.value.trim(); const accountNumber = accountInput.value.trim(); if (!Number.isInteger(amount) || amount < 1000) { toast('Minimum withdrawal is 1,000 Coins.'); return; } if (!paymentMethod) { toast('Please select a payment method.'); return; } if (!accountNumber || accountNumber.length < 5) { toast('Please enter a valid account number.'); return; } try { const data = await api('/withdraw', { method: 'POST', body: JSON.stringify({ amount, paymentMethod, accountNumber }) }); if (!data.success) { toast(data.message || 'Withdrawal failed.'); return; } toast( data.message || 'Withdrawal request submitted successfully.' ); amountInput.value = ''; accountInput.value = ''; await loadWallet(); } catch (error) { console.error(error); toast(error.message || 'Withdrawal failed.'); } }
